@alpha_T_requires : ∀ {enc : Key → Nat} {sp : LegionCore.Spec} {s : LegionCore.State} {r : LegionCore.Req} {rr : RawReq}
  {tasks : List RawTask} {agents : List String} {ledger : List RawEffect} {nonces fenced : List String}
  {uid wid chash mission : String} {pv : Bool} {gv : LegionCore.V},
  AlphaShape enc sp s r rr tasks agents ledger nonces fenced uid wid chash mission pv gv →
    ∀ {ts : List LegionCore.Journal.JTask} {rs : List LegionCore.Journal.Rec},
      decision sp s r ts rs = LegionCore.V.T → pv = true ∧ gv = LegionCore.V.T
@alpha_sound : ∀ {enc : Key → Nat} {sp : LegionCore.Spec} {s : LegionCore.State} {r : LegionCore.Req} {rr : RawReq}
  {tasks : List RawTask} {agents : List String} {ledger : List RawEffect} {nonces fenced : List String}
  {uid wid chash mission : String} {pv : Bool} {gv : LegionCore.V},
  (∀ (a b : Key), enc a = enc b → a = b) →
    AlphaShape enc sp s r rr tasks agents ledger nonces fenced uid wid chash mission pv gv →
      ∀ {ts : List LegionCore.Journal.JTask} {rs : List LegionCore.Journal.Rec},
        decision sp s r ts rs = LegionCore.V.T →
          pv = true ∧
            gv = LegionCore.V.T ∧
              rr.permit.task = rr.task ∧
                rr.permit.cmd = rr.cmd ∧
                  rr.permit.target = rr.target ∧
                    rr.permit.actor = rr.agent ∧
                      rr.agent ∈ agents ∧
                        rr.permit.uid = uid ∧
                          rr.permit.wid = wid ∧
                            rr.permit.mission = mission ∧
                              chash = mission ∧
                                ¬rr.permit.nonce ∈ nonces ∧
                                  ¬rr.cmd ∈ fenced ∧
                                    (∀ (e : RawEffect), e ∈ ledger → e.cmd ≠ rr.cmd ∧ e.task ≠ rr.task) ∧
                                      ∃ t,
                                        t ∈ tasks ∧
                                          t.id = rr.task ∧ t.owner = rr.agent ∧ t.write = true ∧ t.target = rr.target
step_refines : ∀ (x : RStep),
  x.disciplined →
    ∃ c s',
      LegionCore.exec LegionCore.Guards.full x.sp x.pre c = some s' ∧
        s'.ledger = x.newEffects ++ x.pre.ledger ∧
          (x.outcome.creates = true →
              c = LegionCore.Cmd.actuate x.r ∧ s'.usedNonces = x.r.permit.nonce :: x.pre.usedNonces) ∧
            (x.outcome.creates = false → c = LegionCore.Cmd.crash ∧ s' = x.pre)
runtime_effect_admitted : ∀ (x : RStep),
  x.disciplined →
    ∀ {e : LegionCore.Effect},
      e ∈ x.newEffects →
        x.outcome.creates = true ∧
          e = LegionCore.effectOf x.sp x.r ∧ LegionCore.admission LegionCore.Guards.full x.sp x.pre x.r = LegionCore.V.T
trace_effects_admitted : ∀ (tr : List RStep),
  (∀ (x : RStep), x ∈ tr → x.disciplined) →
    ∀ {e : LegionCore.Effect},
      e ∈ appended tr →
        ∃ x,
          x ∈ tr ∧
            x.outcome.creates = true ∧
              e = LegionCore.effectOf x.sp x.r ∧
                LegionCore.admission LegionCore.Guards.full x.sp x.pre x.r = LegionCore.V.T
ledger_continuity : ∀ (tr : List RStep) (x : RStep),
  Continuous (x :: tr) →
    ∀ (y : RStep), y ∈ (x :: tr).getLast? → (y.newEffects ++ y.s.ledger).Perm (appended (x :: tr) ++ x.s.ledger)
trace_once : ∀ (tr : List RStep) (x : RStep),
  (∀ (y : RStep), y ∈ x :: tr → y.disciplined) →
    Continuous (x :: tr) →
      (List.map (fun x => x.task) x.s.ledger).Nodup →
        ∀ (y : RStep), y ∈ x :: tr → (List.map (fun x => x.task) (y.newEffects ++ y.s.ledger)).Nodup
grun_one_create : ∀ (es : List GEv) (g g' : GSt),
  grun g es = some g' → (g.st ≠ GS.reserved → g'.creates = g.creates) ∧ g'.creates ≤ g.creates + 1
grun_one_capture : ∀ (es : List GEv) (g g' : GSt),
  grun g es = some g' → (g.captureStarted = true → g'.captures = g.captures) ∧ g'.captures ≤ g.captures + 1
@dispatch_needs_lean : ∀ {g g' : GSt} {lean : Bool},
  (gstep g (GEv.dispatchCreate lean) = some g' → lean = true) ∧
    (gstep g (GEv.dispatchCapture lean) = some g' → lean = true)
uncertainty_sound : ∀ (es : List GEv) (g' : GSt),
  grun GSt.init es = some g' →
    ∀ (n : Nat), consistentCaptures g' n → n ≤ 1 ∧ (n = 1 → ∃ e, e ∈ es ∧ e = GEv.dispatchCapture true)
